public class IkiIslem {
    public static void main(String[] args) {
        int a = 17;
        int b = 5;
        System.out.println(a + " + " + b + " = " + (a + b));
        System.out.println(a + " - " + b + " = " + (a - b));
        System.out.println(a + " * " + b + " = " + (a * b));
        System.out.println(a + " / " + b + " = " + (a / b));
        System.out.println(a + " % " + b + " = " + (a % b));
        // b degiskeni double yapildiginda bolme islenmi (a/b) tamsayi bolmesinden ondalikli bolmeye donusur ve sonuc 3 yerine 3.4 olur.
    }
    
}
